js/block-factory.js
window.BlockFactory = {
    blocks: [],
    activeCategory: 'output',

    blockPaths: [
        'blocks/output/print/print.json',
        'blocks/variables/var_assign/var_assign.json',
        'blocks/logic/if_condition/if_condition.json'
    ],

    async initPalette() {
        this.bindCategoryButtons();
        await this.loadAllBlockConfigs();
        this.renderCategory(this.activeCategory);
    },

    async loadAllBlockConfigs() {
        this.blocks = [];
        for (const path of this.blockPaths) {
            try {
                const res = await fetch(path);
                const config = await res.json();
                this.blocks.push(config);
            } catch (err) {
                console.warn(`Could not load block config from ${path}:`, err);
            }
        }
    },

    bindCategoryButtons() {
        const buttons = document.querySelectorAll('.category-btn');
        buttons.forEach(btn => {
            btn.addEventListener('click', (e) => {
                const targetBtn = e.currentTarget;
                const category = targetBtn.getAttribute('data-category');

                buttons.forEach(b => b.classList.remove('active'));
                targetBtn.classList.add('active');

                this.renderCategory(category);
            });
        });
    },

    createBlockElement(config, currentValues = {}) {
        const el = document.createElement('div');
        el.className = `block ${config.cssClass || 'block-print'}`;
        el.setAttribute('draggable', 'true');
        el.dataset.id = config.id;

        let templateHtml = config.template ? config.template.display : config.id;

        if (config.template && config.template.inputs) {
            config.template.inputs.forEach(input => {
                const val = currentValues[input.key] !== undefined ? currentValues[input.key] : input.default;
                const inputElement = `<input type="${input.type}" data-key="${input.key}" value='${val}' placeholder="${input.placeholder || ''}" />`;
                templateHtml = templateHtml.replace(`{input:${input.key}}`, inputElement);
            });
        }

        el.innerHTML = templateHtml;

        el.querySelectorAll('input').forEach(input => {
            input.addEventListener('mousedown', (e) => e.stopPropagation());
            input.addEventListener('input', () => {
                if (window.PythonEngine) window.PythonEngine.updatePreview();
            });
        });

        return el;
    },

    appendBlockToWorkspace(config) {
        const workspace = document.getElementById('workspace');
        if (!workspace) return;

        const placeholder = workspace.querySelector('.placeholder');
        if (placeholder) placeholder.remove();

        const newBlock = this.createBlockElement(config);
        
        if (window.bindBlockDragEvents) {
            window.bindBlockDragEvents(newBlock, 'workspace');
        }

        workspace.appendChild(newBlock);

        if (window.PythonEngine) window.PythonEngine.updatePreview();
    },

    renderCategory(group) {
        this.activeCategory = group;
        const drawerContent = document.getElementById('drawer-content');
        const drawerTitle = document.getElementById('drawer-title');

        if (!drawerContent || !drawerTitle) return;

        drawerTitle.innerText = `${group.charAt(0).toUpperCase() + group.slice(1)} Blocks`;
        drawerContent.innerHTML = '';

        const matchingBlocks = this.blocks.filter(b => b.category === group);

        matchingBlocks.forEach(config => {
            const el = this.createBlockElement(config);
            
            if (window.bindBlockDragEvents) {
                window.bindBlockDragEvents(el, 'drawer');
            }

            drawerContent.appendChild(el);
        });
    }
};

function initDragAndDrop() {
    const workspace = document.getElementById('workspace');
    const drawerContent = document.getElementById('drawer-content');
    let ghostBlock = null;
    let currentDraggedBlock = null;
    let sourceContainer = null;
    let lastAfterElement = undefined;

    function getDragAfterElement(container, y) {
        const draggableElements = [...container.querySelectorAll('.block:not(.block-ghost)')];

        return draggableElements.reduce((closest, child) => {
            const box = child.getBoundingClientRect();
            const offset = y - box.top - box.height / 2;

            if (offset < 0 && offset > closest.offset) {
                return { offset: offset, element: child };
            } else {
                return closest;
            }
        }, { offset: Number.NEGATIVE_INFINITY }).element;
    }

    function bindBlockDragEvents(block, origin) {
        block.setAttribute('draggable', 'true');

        let startX = 0;
        let startY = 0;
        let startTime = 0;

        block.addEventListener('mousedown', (e) => {
            startX = e.clientX;
            startY = e.clientY;
            startTime = Date.now();
        });

        // Fixed click handler for adding block directly from drawer
        block.addEventListener('click', (e) => {
            if (e.target.tagName.toLowerCase() === 'input') return;
            
            const isDrawerBlock = block.closest('#drawer-content') !== null;
            if (!isDrawerBlock && origin !== 'drawer') return;

            const deltaX = Math.abs(e.clientX - startX);
            const deltaY = Math.abs(e.clientY - startY);
            const duration = Date.now() - startTime;

            if (deltaX < 5 && deltaY < 5 && duration < 300) {
                const blockId = block.dataset.id;
                const config = window.BlockFactory ? window.BlockFactory.blocks.find(b => b.id === blockId) : null;
                if (config) {
                    window.BlockFactory.appendBlockToWorkspace(config);
                }
            }
        });

        block.addEventListener('dragstart', (e) => {
            currentDraggedBlock = block;
            sourceContainer = origin;

            const blockId = block.dataset.id;
            const inputValues = {};
            
            block.querySelectorAll('input').forEach(input => {
                inputValues[input.dataset.key] = input.value;
            });

            e.dataTransfer.setData('text/plain', JSON.stringify({
                id: blockId,
                values: inputValues,
                origin: origin
            }));

            if (origin === 'workspace') {
                setTimeout(() => block.style.opacity = '0.3', 0);
            }
        });

        block.addEventListener('dragend', () => {
            if (currentDraggedBlock) {
                currentDraggedBlock.style.opacity = '1';
            }
            currentDraggedBlock = null;
            sourceContainer = null;
            lastAfterElement = undefined;

            if (ghostBlock) {
                ghostBlock.remove();
                ghostBlock = null;
            }
        });
    }

    if (drawerContent && !drawerContent.dataset.dragAttached) {
        drawerContent.dataset.dragAttached = 'true';

        drawerContent.addEventListener('dragover', (e) => {
            if (sourceContainer === 'workspace') {
                e.preventDefault();
            }
        });

        drawerContent.addEventListener('drop', (e) => {
            e.preventDefault();
            const rawData = e.dataTransfer.getData('text/plain');
            if (!rawData) return;

            const data = JSON.parse(rawData);

            if (data.origin === 'workspace' && currentDraggedBlock) {
                currentDraggedBlock.remove();
                
                if (workspace.querySelectorAll('.block').length === 0) {
                    workspace.innerHTML = '<p class="placeholder">Drag blocks here to build code...</p>';
                }

                if (window.PythonEngine) window.PythonEngine.updatePreview();
            }
        });
    }

    if (workspace && !workspace.dataset.listenersAttached) {
        workspace.dataset.listenersAttached = 'true';

        workspace.addEventListener('dragover', (e) => {
            e.preventDefault();

            if (!ghostBlock && (currentDraggedBlock || e.dataTransfer.types.includes('text/plain'))) {
                if (currentDraggedBlock) {
                    ghostBlock = currentDraggedBlock.cloneNode(true);
                    const originalInputs = currentDraggedBlock.querySelectorAll('input');
                    const ghostInputs = ghostBlock.querySelectorAll('input');
                    originalInputs.forEach((inp, idx) => {
                        if (ghostInputs[idx]) ghostInputs[idx].value = inp.value;
                    });
                } else {
                    ghostBlock = document.createElement('div');
                    ghostBlock.className = 'block block-print';
                    ghostBlock.innerText = 'Drop here...';
                }
                ghostBlock.classList.add('block-ghost');
            }

            if (ghostBlock) {
                const afterElement = getDragAfterElement(workspace, e.clientY);

                if (afterElement !== lastAfterElement) {
                    lastAfterElement = afterElement;
                    if (afterElement == null) {
                        workspace.appendChild(ghostBlock);
                    } else {
                        workspace.insertBefore(ghostBlock, afterElement);
                    }
                }
            }
        });

        workspace.addEventListener('dragleave', (e) => {
            if (e.target === workspace && ghostBlock && ghostBlock.parentNode) {
                const rect = workspace.getBoundingClientRect();
                if (e.clientX < rect.left || e.clientX > rect.right || e.clientY < rect.top || e.clientY > rect.bottom) {
                    ghostBlock.remove();
                    ghostBlock = null;
                    lastAfterElement = undefined;
                }
            }
        });

        workspace.addEventListener('drop', (e) => {
            e.preventDefault();
            const rawData = e.dataTransfer.getData('text/plain');
            if (!rawData) return;

            let blockData = JSON.parse(rawData);
            const placeholder = workspace.querySelector('.placeholder');
            if (placeholder) placeholder.remove();

            const targetPosition = lastAfterElement;

            if (blockData.origin === 'workspace' && currentDraggedBlock) {
                if (targetPosition) {
                    workspace.insertBefore(currentDraggedBlock, targetPosition);
                } else {
                    workspace.appendChild(currentDraggedBlock);
                }
            } else {
                const config = window.BlockFactory.blocks.find(b => b.id === blockData.id);
                if (config) {
                    const newBlock = window.BlockFactory.createBlockElement(config, blockData.values);
                    
                    bindBlockDragEvents(newBlock, 'workspace');

                    if (targetPosition) {
                        workspace.insertBefore(newBlock, targetPosition);
                    } else {
                        workspace.appendChild(newBlock);
                    }
                }
            }

            if (ghostBlock) {
                ghostBlock.remove();
                ghostBlock = null;
            }
            lastAfterElement = undefined;

            if (window.PythonEngine) window.PythonEngine.updatePreview();
        });
    }

    window.bindBlockDragEvents = bindBlockDragEvents;
}

window.PythonEngine = {
    async init() {
        this.bindButtons();
        this.updatePreview();
    },

    bindButtons() {
        const runBtn = document.getElementById('run-btn');
        const exportBtn = document.getElementById('export-btn');

        if (runBtn) {
            runBtn.addEventListener('click', () => this.runCode());
        }

        if (exportBtn) {
            exportBtn.addEventListener('click', () => this.exportCode());
        }
    },

    generatePythonCode() {
        const workspace = document.getElementById('workspace');
        if (!workspace) return '# No workspace found';

        const blocks = workspace.querySelectorAll('.block:not(.block-ghost)');
        if (blocks.length === 0) {
            return '# No blocks in workspace';
        }

        let codeLines = [];

        blocks.forEach(block => {
            const blockId = block.dataset.id;
            const config = window.BlockFactory ? window.BlockFactory.blocks.find(b => b.id === blockId) : null;

            if (config && config.codeMapping && config.codeMapping.python) {
                let line = config.codeMapping.python;
                
                block.querySelectorAll('input').forEach(input => {
                    const key = input.dataset.key;
                    const val = input.value;
                    line = line.replace(`{input:${key}}`, val);
                });

                codeLines.push(line);
            } else {
                if (blockId === 'print') {
                    const input = block.querySelector('input');
                    const val = input ? input.value : '';
                    codeLines.push(`print(${val})`);
                } else {
                    codeLines.push(`# ${blockId} block execution`);
                }
            }
        });

        return codeLines.join('\n');
    },

    updatePreview() {
        const previewEl = document.getElementById('python-preview');
        const code = this.generatePythonCode();

        if (previewEl) {
            previewEl.textContent = `# Generated by PyBlocks Studio\n\n${code}`;
        }
    },

    async runCode() {
        const consoleOutput = document.getElementById('console-output');
        const code = this.generatePythonCode();

        if (!consoleOutput) return;

        consoleOutput.innerHTML = `<p class="console-line">&gt; Executing Python script...</p>`;

        try {
            let outputBuffer = [];
            let scopeVars = {};

            const lines = code.split('\n');
            lines.forEach(line => {
                line = line.trim();
                if (!line || line.startsWith('#')) return;

                // Handle Variable Assignments: x = 10 or x = "hello"
                if (line.includes('=')) {
                    const parts = line.split('=');
                    const varName = parts[0].trim();
                    let varVal = parts[1].trim();

                    try {
                        scopeVars[varName] = eval(varVal);
                    } catch (e) {
                        scopeVars[varName] = varVal;
                    }
                } 
                // Handle Print Calls: print(...)
                else if (line.startsWith('print(') && line.endsWith(')')) {
                    const inner = line.slice(6, -1).trim();
                    
                    if ((inner.startsWith('"') && inner.endsWith('"')) || (inner.startsWith("'") && inner.endsWith("'"))) {
                        outputBuffer.push(inner.slice(1, -1));
                    } else if (scopeVars.hasOwnProperty(inner)) {
                        outputBuffer.push(scopeVars[inner]);
                    } else if (!isNaN(inner) && inner !== '') {
                        outputBuffer.push(Number(inner));
                    } else {
                        outputBuffer.push(inner);
                    }
                }
            });

            if (outputBuffer.length > 0) {
                outputBuffer.forEach(line => {
                    consoleOutput.innerHTML += `<p class="console-line">&gt; [STDOUT]: ${line}</p>`;
                });
            } else {
                consoleOutput.innerHTML += `<p class="console-line text-muted">&gt; (Script executed with no output)</p>`;
            }

            consoleOutput.innerHTML += `<p class="console-line">&gt; Process finished with code 0</p>`;
        } catch (err) {
